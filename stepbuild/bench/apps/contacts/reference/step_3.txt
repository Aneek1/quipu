"""API tests for the contacts app, using the client fixture from conftest.py."""


def create(client, **body):
    body.setdefault("email", "ada@example.com")
    response = client.post("/api/contacts", json=body)
    assert response.status_code == 201
    return response.get_json()


def test_create_returns_201_and_the_contact(client):
    contact = create(client, name="Ada")
    assert isinstance(contact["id"], int)
    assert contact["name"] == "Ada"
    assert contact["email"] == "ada@example.com"
    assert contact["phone"] == ""


def test_list_returns_a_json_array(client):
    assert client.get("/api/contacts").get_json() == []
    contact = create(client, name="Ada", phone="555-0100")
    response = client.get("/api/contacts")
    assert response.status_code == 200
    assert response.get_json() == [contact]


def test_get_one(client):
    contact = create(client, name="Ada")
    response = client.get(f"/api/contacts/{contact['id']}")
    assert response.status_code == 200
    assert response.get_json() == contact


def test_update(client):
    contact = create(client, name="Ada")
    response = client.put(f"/api/contacts/{contact['id']}", json={"phone": "555-0101"})
    assert response.status_code == 200
    assert response.get_json() == {**contact, "phone": "555-0101"}


def test_update_rejects_a_bad_email(client):
    contact = create(client, name="Ada")
    response = client.put(f"/api/contacts/{contact['id']}", json={"email": "nope"})
    assert response.status_code == 400
    assert "error" in response.get_json()


def test_delete_then_404(client):
    contact = create(client, name="Ada")
    assert client.delete(f"/api/contacts/{contact['id']}").status_code == 204
    assert client.get(f"/api/contacts/{contact['id']}").status_code == 404
    assert client.delete(f"/api/contacts/{contact['id']}").status_code == 404


def test_missing_name_is_400(client):
    response = client.post("/api/contacts", json={"email": "a@b.c"})
    assert response.status_code == 400
    assert "error" in response.get_json()


def test_bad_emails_are_400(client):
    for email in (None, "", "ada", "@example.com", "ada@", "a@b@c"):
        body = {"name": "Ada", "email": email}
        assert client.post("/api/contacts", json=body).status_code == 400, email


def test_ids_are_never_reused(client):
    create(client, name="A")
    b = create(client, name="B")
    client.delete(f"/api/contacts/{b['id']}")
    assert create(client, name="C")["id"] != b["id"]
