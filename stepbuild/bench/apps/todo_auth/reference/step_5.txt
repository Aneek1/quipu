async function request(path, options = {}) {
  const response = await fetch(path, {
    ...options,
    credentials: 'same-origin',
    headers: { 'Content-Type': 'application/json' },
  })
  if (!response.ok) {
    const err = new Error(`${options.method || 'GET'} ${path} failed with status ${response.status}`)
    err.status = response.status
    throw err
  }
  if (response.status === 204) {
    return null
  }
  return response.json()
}

export async function register(username, password) {
  const body = JSON.stringify({ username, password })
  return request('/api/auth/register', { method: 'POST', body })
}

export async function login(username, password) {
  const body = JSON.stringify({ username, password })
  return request('/api/auth/login', { method: 'POST', body })
}

export async function logout() {
  return request('/api/auth/logout', { method: 'POST' })
}

export async function listTodos() {
  return request('/api/todos')
}

export async function createTodo(todo) {
  return request('/api/todos', { method: 'POST', body: JSON.stringify(todo) })
}

export async function deleteTodo(id) {
  return request(`/api/todos/${id}`, { method: 'DELETE' })
}

import { useEffect, useState } from 'react'
import List from './components/List.jsx'
import Form from './components/Form.jsx'
import { createTodo, deleteTodo, listTodos, login, logout, register } from './api.js'

export default function App() {
  const [items, setItems] = useState([])
  const [loggedIn, setLoggedIn] = useState(false)
  const [username, setUsername] = useState('')
  const [password, setPassword] = useState('')
  const [error, setError] = useState(null)

  async function load() {
    try {
      setItems(await listTodos())
      setLoggedIn(true)
      setError(null)
    } catch (err) {
      setLoggedIn(false)
      if (err.status !== 401) {
        setError(err.message)
      }
    }
  }

  useEffect(() => {
    load()
  }, [])

  async function run(action) {
    try {
      await action()
      setError(null)
    } catch (err) {
      setError(err.message)
    }
  }

  function handleLogin(event) {
    event.preventDefault()
    run(async () => {
      await login(username, password)
      setPassword('')
      await load()
    })
  }

  const handleRegister = () => run(() => register(username, password))
  const handleLogout = () =>
    run(async () => {
      await logout()
      setLoggedIn(false)
      setItems([])
    })
  const handleCreate = (values) =>
    run(async () => {
      const todo = await createTodo(values)
      setItems((current) => [...current, todo])
    })
  const handleDelete = (id) =>
    run(async () => {
      await deleteTodo(id)
      setItems((current) => current.filter((item) => item.id !== id))
    })

  return (
    <main>
      <h1>Todos</h1>
      {error && <p role="alert">{error}</p>}
      {loggedIn ? (
        <>
          <button type="button" onClick={handleLogout}>
            Log out
          </button>
          <Form onSubmit={handleCreate} />
          <List items={items} onDelete={handleDelete} />
        </>
      ) : (
        <form onSubmit={handleLogin}>
          <input
            value={username}
            onChange={(event) => setUsername(event.target.value)}
            placeholder="Username"
          />
          <input
            type="password"
            value={password}
            onChange={(event) => setPassword(event.target.value)}
            placeholder="Password"
          />
          <button type="submit">Log in</button>
          <button type="button" onClick={handleRegister}>
            Register
          </button>
        </form>
      )}
    </main>
  )
}
