async function request(path, options = {}) {
  const response = await fetch(path, {
    ...options,
    headers: { 'Content-Type': 'application/json' },
  })
  if (!response.ok) {
    throw new Error(`${options.method || 'GET'} ${path} failed with status ${response.status}`)
  }
  if (response.status === 204) {
    return null
  }
  return response.json()
}

export async function listBookmarks() {
  return request('/api/bookmarks')
}

export async function createBookmark(bookmark) {
  return request('/api/bookmarks', { method: 'POST', body: JSON.stringify(bookmark) })
}

export async function updateBookmark(id, changes) {
  return request(`/api/bookmarks/${id}`, { method: 'PUT', body: JSON.stringify(changes) })
}

export async function deleteBookmark(id) {
  return request(`/api/bookmarks/${id}`, { method: 'DELETE' })
}

import { useEffect, useState } from 'react'
import List from './components/List.jsx'
import Form from './components/Form.jsx'
import { createBookmark, deleteBookmark, listBookmarks } from './api.js'

export default function App() {
  const [items, setItems] = useState([])
  const [error, setError] = useState(null)

  useEffect(() => {
    listBookmarks()
      .then(setItems)
      .catch((err) => setError(err.message))
  }, [])

  async function handleCreate(values) {
    try {
      const bookmark = await createBookmark(values)
      setItems((current) => [...current, bookmark])
      setError(null)
    } catch (err) {
      setError(err.message)
    }
  }

  async function handleDelete(id) {
    try {
      await deleteBookmark(id)
      setItems((current) => current.filter((item) => item.id !== id))
      setError(null)
    } catch (err) {
      setError(err.message)
    }
  }

  return (
    <main>
      <h1>Bookmarks</h1>
      {error && <p role="alert">{error}</p>}
      <Form onSubmit={handleCreate} />
      <List items={items} onDelete={handleDelete} />
    </main>
  )
}
