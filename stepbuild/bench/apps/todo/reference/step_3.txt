"""API tests for the todo app, using the client fixture from conftest.py."""


def create(client, **body):
    response = client.post("/api/todos", json=body)
    assert response.status_code == 201
    return response.get_json()


def test_create_returns_201_and_the_todo(client):
    todo = create(client, title="Buy milk")
    assert isinstance(todo["id"], int)
    assert todo["title"] == "Buy milk"
    assert todo["done"] is False


def test_list_returns_a_json_array(client):
    assert client.get("/api/todos").get_json() == []
    todo = create(client, title="Buy milk")
    response = client.get("/api/todos")
    assert response.status_code == 200
    assert response.get_json() == [todo]


def test_get_one(client):
    todo = create(client, title="Buy milk")
    response = client.get(f"/api/todos/{todo['id']}")
    assert response.status_code == 200
    assert response.get_json() == todo


def test_update(client):
    todo = create(client, title="Buy milk")
    response = client.put(f"/api/todos/{todo['id']}", json={"done": True})
    assert response.status_code == 200
    assert response.get_json() == {"id": todo["id"], "title": "Buy milk", "done": True}


def test_update_rejects_an_empty_title(client):
    todo = create(client, title="Buy milk")
    response = client.put(f"/api/todos/{todo['id']}", json={"title": ""})
    assert response.status_code == 400
    assert "error" in response.get_json()


def test_delete_then_404(client):
    todo = create(client, title="Buy milk")
    assert client.delete(f"/api/todos/{todo['id']}").status_code == 204
    assert client.get(f"/api/todos/{todo['id']}").status_code == 404
    assert client.delete(f"/api/todos/{todo['id']}").status_code == 404


def test_missing_title_is_400(client):
    response = client.post("/api/todos", json={"done": False})
    assert response.status_code == 400
    assert "error" in response.get_json()


def test_non_boolean_done_is_400(client):
    response = client.post("/api/todos", json={"title": "Buy milk", "done": "yes"})
    assert response.status_code == 400


def test_missing_todo_is_404(client):
    assert client.get("/api/todos/999").status_code == 404
    assert client.put("/api/todos/999", json={"title": "x"}).status_code == 404


def test_ids_are_never_reused(client):
    create(client, title="A")
    b = create(client, title="B")
    client.delete(f"/api/todos/{b['id']}")
    assert create(client, title="C")["id"] != b["id"]


def test_blank_title_is_400(client):
    assert client.post("/api/todos", json={"title": "   "}).status_code == 400
