"""In-memory storage and validation for the recipes app."""


class Store:
    """Keeps items in a dict keyed by an integer id assigned on create.

    Ids come from a counter, never from the number of items, so an id is never
    reused after a delete. get and update return None and delete returns False
    for an id that does not exist; delete returns True when it removed the item.
    """

    def __init__(self):
        self._items = {}
        self._next_id = 1

    def create(self, data):
        item = dict(data)
        item["id"] = self._next_id
        self._items[item["id"]] = item
        self._next_id += 1
        return dict(item)

    def list_items(self):
        return [dict(item) for item in self._items.values()]

    def get(self, id):
        item = self._items.get(id)
        return dict(item) if item is not None else None

    def update(self, id, data):
        item = self._items.get(id)
        if item is None:
            return None
        item.update(data)
        item["id"] = id
        return dict(item)

    def delete(self, id):
        return self._items.pop(id, None) is not None


def validate_recipe(data):
    """Return a list of error strings; empty when `data` is a valid recipe.

    bool is a subclass of int in Python, so `minutes` rejects booleans explicitly.
    """
    if not isinstance(data, dict):
        return ["the request body must be a JSON object"]
    errors = []
    title = data.get("title")
    if not isinstance(title, str) or not title.strip():
        errors.append("title is required and must be a non-empty string")
    if "ingredients" in data:
        ingredients = data["ingredients"]
        if not isinstance(ingredients, list) or not all(
            isinstance(i, str) and i.strip() for i in ingredients
        ):
            errors.append("ingredients must be a list of non-empty strings")
    if "minutes" in data:
        minutes = data["minutes"]
        if not isinstance(minutes, int) or isinstance(minutes, bool):
            errors.append("minutes must be an integer")
        elif minutes < 0:
            errors.append("minutes must be 0 or more")
    return errors
