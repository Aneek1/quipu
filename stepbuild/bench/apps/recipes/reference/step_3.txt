"""API tests for the recipes app, using the client fixture from conftest.py."""


def create(client, **body):
    response = client.post("/api/recipes", json=body)
    assert response.status_code == 201
    return response.get_json()


def test_create_returns_201_and_the_recipe(client):
    recipe = create(client, title="Pancakes")
    assert isinstance(recipe["id"], int)
    assert recipe["title"] == "Pancakes"
    assert recipe["ingredients"] == []
    assert recipe["minutes"] == 0


def test_list_returns_a_json_array(client):
    assert client.get("/api/recipes").get_json() == []
    recipe = create(client, title="Pancakes", ingredients=["flour", "milk"], minutes=20)
    response = client.get("/api/recipes")
    assert response.status_code == 200
    assert response.get_json() == [recipe]


def test_get_one(client):
    recipe = create(client, title="Pancakes")
    response = client.get(f"/api/recipes/{recipe['id']}")
    assert response.status_code == 200
    assert response.get_json() == recipe


def test_update(client):
    recipe = create(client, title="Pancakes", minutes=20)
    response = client.put(f"/api/recipes/{recipe['id']}", json={"minutes": 25})
    assert response.status_code == 200
    assert response.get_json() == {**recipe, "minutes": 25}


def test_update_rejects_negative_minutes(client):
    recipe = create(client, title="Pancakes")
    response = client.put(f"/api/recipes/{recipe['id']}", json={"minutes": -1})
    assert response.status_code == 400
    assert "error" in response.get_json()


def test_delete_then_404(client):
    recipe = create(client, title="Pancakes")
    assert client.delete(f"/api/recipes/{recipe['id']}").status_code == 204
    assert client.get(f"/api/recipes/{recipe['id']}").status_code == 404
    assert client.delete(f"/api/recipes/{recipe['id']}").status_code == 404


def test_missing_title_is_400(client):
    response = client.post("/api/recipes", json={"minutes": 5})
    assert response.status_code == 400
    assert "error" in response.get_json()


def test_bad_ingredients_are_400(client):
    assert client.post("/api/recipes", json={"title": "x", "ingredients": "flour"}).status_code == 400
    assert client.post("/api/recipes", json={"title": "x", "ingredients": [""]}).status_code == 400


def test_bad_minutes_are_400(client):
    assert client.post("/api/recipes", json={"title": "x", "minutes": 1.5}).status_code == 400
    assert client.post("/api/recipes", json={"title": "x", "minutes": True}).status_code == 400


def test_ids_are_never_reused(client):
    create(client, title="A")
    b = create(client, title="B")
    client.delete(f"/api/recipes/{b['id']}")
    assert create(client, title="C")["id"] != b["id"]
