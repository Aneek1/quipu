async function request(path, options = {}) {
  const response = await fetch(path, {
    ...options,
    headers: { 'Content-Type': 'application/json' },
  })
  if (!response.ok) {
    throw new Error(`${options.method || 'GET'} ${path} failed with status ${response.status}`)
  }
  if (response.status === 204) {
    return null
  }
  return response.json()
}

export async function listHabits() {
  return request('/api/habits')
}

export async function createHabit(habit) {
  return request('/api/habits', { method: 'POST', body: JSON.stringify(habit) })
}

export async function deleteHabit(id) {
  return request(`/api/habits/${id}`, { method: 'DELETE' })
}

export async function checkIn(id, date) {
  return request(`/api/habits/${id}/checkins`, { method: 'POST', body: JSON.stringify({ date }) })
}

export async function listCheckins(id) {
  return request(`/api/habits/${id}/checkins`)
}

import { useEffect, useState } from 'react'
import List from './components/List.jsx'
import Form from './components/Form.jsx'
import { createHabit, deleteHabit, listHabits } from './api.js'

export default function App() {
  const [items, setItems] = useState([])
  const [error, setError] = useState(null)

  useEffect(() => {
    listHabits()
      .then(setItems)
      .catch((err) => setError(err.message))
  }, [])

  async function handleCreate(values) {
    try {
      const habit = await createHabit(values)
      setItems((current) => [...current, habit])
      setError(null)
    } catch (err) {
      setError(err.message)
    }
  }

  async function handleDelete(id) {
    try {
      await deleteHabit(id)
      setItems((current) => current.filter((item) => item.id !== id))
      setError(null)
    } catch (err) {
      setError(err.message)
    }
  }

  return (
    <main>
      <h1>Habits</h1>
      {error && <p role="alert">{error}</p>}
      <Form onSubmit={handleCreate} />
      <List items={items} onDelete={handleDelete} />
    </main>
  )
}
