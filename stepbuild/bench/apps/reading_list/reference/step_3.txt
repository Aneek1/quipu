"""API tests for the reading_list app, using the client fixture from conftest.py."""


def create(client, **body):
    body.setdefault("author", "Ursula K. Le Guin")
    response = client.post("/api/books", json=body)
    assert response.status_code == 201
    return response.get_json()


def test_create_returns_201_and_the_book(client):
    book = create(client, title="The Dispossessed")
    assert isinstance(book["id"], int)
    assert book["title"] == "The Dispossessed"
    assert book["author"] == "Ursula K. Le Guin"
    assert book["status"] == "to-read"


def test_list_returns_a_json_array(client):
    assert client.get("/api/books").get_json() == []
    book = create(client, title="Earthsea", status="reading")
    response = client.get("/api/books")
    assert response.status_code == 200
    assert response.get_json() == [book]


def test_get_one(client):
    book = create(client, title="Earthsea")
    response = client.get(f"/api/books/{book['id']}")
    assert response.status_code == 200
    assert response.get_json() == book


def test_update(client):
    book = create(client, title="Earthsea")
    response = client.put(f"/api/books/{book['id']}", json={"status": "done"})
    assert response.status_code == 200
    assert response.get_json() == {**book, "status": "done"}


def test_update_rejects_an_unknown_status(client):
    book = create(client, title="Earthsea")
    response = client.put(f"/api/books/{book['id']}", json={"status": "finished"})
    assert response.status_code == 400
    assert "error" in response.get_json()


def test_delete_then_404(client):
    book = create(client, title="Earthsea")
    assert client.delete(f"/api/books/{book['id']}").status_code == 204
    assert client.get(f"/api/books/{book['id']}").status_code == 404
    assert client.delete(f"/api/books/{book['id']}").status_code == 404


def test_missing_title_is_400(client):
    response = client.post("/api/books", json={"author": "Someone"})
    assert response.status_code == 400
    assert "error" in response.get_json()


def test_missing_author_is_400(client):
    assert client.post("/api/books", json={"title": "Earthsea"}).status_code == 400


def test_ids_are_never_reused(client):
    create(client, title="A")
    b = create(client, title="B")
    client.delete(f"/api/books/{b['id']}")
    assert create(client, title="C")["id"] != b["id"]
